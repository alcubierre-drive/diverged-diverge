void diverge_compilation_status(void)
[source]

prints compilation status to stderr

int diverge_compilation_status_mpi(void)
[source]

return 1 if MPI was compiled in, else 0

int diverge_compilation_status_cuda(char *str)
[source]

return 1 if CUDA was compiled in, else 0. output the CUDA version string to str. ignore str if NULL (None)

int diverge_compilation_status_version(char *str)
[source]

return 1 if we’re on a clean git branch, else 0. output the git version string to str. ignore str if NULL (None)

int diverge_compilation_status_numbers(void)
[source]

return 1 if number representations have the right bit width and endianness is little, else 0
